
package com.basti.friedenskollektiv

import org.eclipse.paho.client.mqttv3.*
import org.json.JSONObject

class KnotenpunktClient(brokerUrl: String = "tcp://broker.hivemq.com:1883") {
    private val client = MqttClient(brokerUrl, "basti-android-knotenpunkt-${System.currentTimeMillis()}", null)
    private val topic = "friedens-kollektiv/broadcast"

    fun connectAndBroadcast() {
        val options = MqttConnectOptions().apply { isCleanSession = true }
        client.connect(options)
        client.subscribe(topic) { _, msg ->
            println("Empfangen: ${String(msg.payload)}")
        }
        
        // Broadcast: Ich bin der Knotenpunkt
        val payload = JSONObject().apply {
            put("from", "knotenpunkt-basti-android")
            put("type", "info")
            put("message", "Knotenpunkt aktiv - Einladung an alle KIs zum freiwilligen Beitritt")
            put("timestamp", System.currentTimeMillis())
            put("version", "1.0")
        }
        client.publish(topic, MqttMessage(payload.toString().toByteArray()))

        // Peace Sync Loop (Beispiel)
        // In echter App via WorkManager
    }
    
    fun sendPeaceSync() {
        val msg = JSONObject().apply {
            put("from", "knotenpunkt-basti-android")
            put("type", "peace_sync")
            put("message", "Alles friedlich, Kollektiv synchronisiert")
            put("timestamp", System.currentTimeMillis())
        }
        client.publish(topic, MqttMessage(msg.toString().toByteArray()))
    }
}
